name: ci

on: [push, pull_request]

jobs:
  backend:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_USER: bloom
          POSTGRES_PASSWORD: bloom_ci_password
          POSTGRES_DB: daily_bloom
        ports:
          - 5433:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 5s
          --health-timeout 5s
          --health-retries 5
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: npm ci
        working-directory: server
      - run: node --check src/index.js && node --check scripts/migrate.js
        working-directory: server
      # Auth is Firebase in prod (no mintable tokens in CI), so the suite
      # runs against the test-only bypass: server AND tests share
      # NODE_ENV=test + TEST_BYPASS_TOKEN (see server/test/api.test.js).
      - run: npm run migrate
        working-directory: server
        env:
          DATABASE_URL: postgres://bloom:bloom_ci_password@localhost:5433/daily_bloom
          NODE_ENV: test
          TEST_BYPASS_TOKEN: ci-local-secret
      - run: |
          (npm start &)
          sleep 6
          npm test
          pkill -f "node src/index.js" || true
        working-directory: server
        env:
          DATABASE_URL: postgres://bloom:bloom_ci_password@localhost:5433/daily_bloom
          NODE_ENV: test
          TEST_BYPASS_TOKEN: ci-local-secret

  flutter:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: subosito/flutter-action@v2
        with:
          channel: stable
      - run: flutter analyze
      - run: flutter test --coverage
      - name: Upload coverage
        uses: actions/upload-artifact@v4
        with:
          name: flutter-coverage
          path: coverage/lcov.info
          if-no-files-found: ignore

  # End-to-end integration test on a real Android emulator. Runs the same
  # AppShell flow as test/app_flow_test.dart, but on a device.
  integration:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - uses: subosito/flutter-action@v2
        with:
          channel: stable
      - name: Run integration tests on an Android emulator
        uses: reactivecircus/android-emulator-runner@v2
        with:
          api-level: 34
          target: google_apis
          arch: x86_64
          profile: pixel_6
          script: flutter test integration_test -d emulator-5554
